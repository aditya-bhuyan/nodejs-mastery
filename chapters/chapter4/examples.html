Chapter 4 — Code Examples

Example 1 — Reading a JSON Config File

// read-config.js
const fsp  = require("fs/promises");
const path = require("path");

async function loadConfig(filename) {
  const filePath = path.join(__dirname, filename);
  const raw = await fsp.readFile(filePath, "utf8");
  return JSON.parse(raw);
}

(async () => {
  const cfg = await loadConfig("package.json");
  console.log("Project:", cfg.name, "v" + cfg.version);
})();

Example 2 — Recursive Directory Listing

// ls-recursive.js
const fsp  = require("fs/promises");
const path = require("path");

async function listDir(dir, indent = 0) {
  const entries = await fsp.readdir(dir, { withFileTypes: true });
  for (const entry of entries) {
    console.log(" ".repeat(indent * 2) + (entry.isDirectory() ? "📁 " : "📄 ") + entry.name);
    if (entry.isDirectory()) {
      await listDir(path.join(dir, entry.name), indent + 1);
    }
  }
}

listDir(".");

Example 3 — Line-by-Line File Reader

// readline-demo.js
const fs       = require("fs");
const readline = require("readline");

const rl = readline.createInterface({
  input: fs.createReadStream("./package.json"),
});

let lineNum = 0;
rl.on("line", (line) => {
  lineNum++;
  console.log(`${String(lineNum).padStart(3)}: ${line}`);
});
rl.on("close", () => console.log(`Total lines: ${lineNum}`));

Example 4 — Streaming File Copy with Progress

// copy-progress.js
const fs   = require("fs");
const path = require("path");
const { pipeline } = require("stream/promises");
const { Transform } = require("stream");

async function copyWithProgress(src, dest) {
  const { size } = await require("fs/promises").stat(src);
  let copied = 0;

  const progress = new Transform({
    transform(chunk, _enc, cb) {
      copied += chunk.length;
      const pct = ((copied / size) * 100).toFixed(1);
      process.stdout.write(`\rCopying... ${pct}%`);
      cb(null, chunk);
    }
  });

  await pipeline(
    fs.createReadStream(src),
    progress,
    fs.createWriteStream(dest)
  );
  console.log("\nDone!");
}

copyWithProgress("package.json", "package.json.bak");

Example 5 — Custom Transform Stream (CSV → JSON)

// csv-to-json.js
const { Transform } = require("stream");

class CsvToJson extends Transform {
  constructor() {
    super({ readableObjectMode: true });
    this._headers = null;
    this._buffer  = "";
  }
  _transform(chunk, _enc, cb) {
    this._buffer += chunk.toString();
    const lines = this._buffer.split("\n");
    this._buffer = lines.pop(); // keep incomplete last line
    for (const line of lines) {
      if (!line.trim()) continue;
      const vals = line.split(",").map(v => v.trim());
      if (!this._headers) { this._headers = vals; continue; }
      const obj = Object.fromEntries(this._headers.map((h, i) => [h, vals[i]]));
      this.push(JSON.stringify(obj) + "\n");
    }
    cb();
  }
  _flush(cb) { cb(); }
}

// Usage: pipe stdin CSV → stdout JSON lines
process.stdin.pipe(new CsvToJson()).pipe(process.stdout);
