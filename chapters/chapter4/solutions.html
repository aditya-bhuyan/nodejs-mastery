Solution 1 — File Stats Reporter

Show solution 
// stats.js
const fsp  = require("fs/promises");
const path = require("path");

async function main() {
  const input = process.argv[2];
  if (!input) { console.error("Usage: node stats.js "); process.exit(1); }

  const abs  = path.resolve(input);
  const stat = await fsp.stat(abs);

  console.log("Name       :", path.basename(abs));
  console.log("Extension  :", path.extname(abs) || "(none)");
  console.log("Size       :", (stat.size / 1024).toFixed(2), "KB");
  console.log("Created    :", stat.birthtime.toLocaleString());
  console.log("Modified   :", stat.mtime.toLocaleString());
  console.log("Type       :", stat.isDirectory() ? "directory" : "file");
}
main();

Solution 2 — Recursive File Search

Show solution 
// find.js
const fsp  = require("fs/promises");
const path = require("path");

async function find(dir, ext, results = []) {
  const entries = await fsp.readdir(dir, { withFileTypes: true });
  for (const e of entries) {
    const full = path.join(dir, e.name);
    if (e.isDirectory()) {
      await find(full, ext, results);
    } else if (e.name.endsWith(ext)) {
      results.push(full);
    }
  }
  return results;
}

(async () => {
  const [dir, ext] = process.argv.slice(2);
  if (!dir || !ext) { console.error("Usage: node find.js  "); process.exit(1); }
  const files = await find(path.resolve(dir), ext);
  files.forEach(f => console.log(f));
  console.log(`\nFound ${files.length} file(s).`);
})();

Solution 3 — Gzip Compressor/Decompressor

Show solution 
// gzip.js
const fs   = require("fs");
const zlib = require("zlib");
const path = require("path");
const { pipeline } = require("stream/promises");

async function main() {
  const [flag, src, dest] = process.argv.slice(2);
  if (!flag || !src || !dest) {
    console.error("Usage: node gzip.js --compress|--decompress  ");
    process.exit(1);
  }
  const transform = flag === "--compress" ? zlib.createGzip() : zlib.createGunzip();
  await pipeline(
    fs.createReadStream(path.resolve(src)),
    transform,
    fs.createWriteStream(path.resolve(dest))
  );
  console.log("Done:", src, "→", dest);
}
main();

Solution 4 — CSV Line Counter

Show solution 
// csvcount.js
const fs       = require("fs");
const readline = require("readline");
const path     = require("path");

function countRows(filePath) {
  return new Promise((resolve, reject) => {
    const rl = readline.createInterface({
      input: fs.createReadStream(path.resolve(filePath)),
    });
    let rows = -1; // skip header
    rl.on("line", () => rows++);
    rl.on("close", () => resolve(rows));
    rl.on("error", reject);
  });
}

(async () => {
  const file = process.argv[2];
  if (!file) { console.error("Usage: node csvcount.js "); process.exit(1); }
  const count = await countRows(file);
  console.log(`Data rows: ${count}`);
})();
