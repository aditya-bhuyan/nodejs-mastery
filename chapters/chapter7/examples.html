Chapter 7 — Code Examples

Example 1 — SQLite Users CRUD

// sqlite-demo.js — install: npm install better-sqlite3
const Database = require("better-sqlite3");
const db = new Database(":memory:"); // in-memory for demo

db.exec(`CREATE TABLE users (
  id    INTEGER PRIMARY KEY AUTOINCREMENT,
  name  TEXT NOT NULL,
  email TEXT NOT NULL UNIQUE
)`);

const stmtInsert = db.prepare("INSERT INTO users (name, email) VALUES (?, ?)");
const stmtAll    = db.prepare("SELECT * FROM users");
const stmtOne    = db.prepare("SELECT * FROM users WHERE id = ?");
const stmtDelete = db.prepare("DELETE FROM users WHERE id = ?");

stmtInsert.run("Alice", "alice@example.com");
stmtInsert.run("Bob",   "bob@example.com");

console.log("All users:", stmtAll.all());
console.log("User 1:",    stmtOne.get(1));

stmtDelete.run(2);
console.log("After delete:", stmtAll.all());

Example 2 — Express + SQLite REST API

// express-sqlite.js
// Install: npm install express better-sqlite3
const express  = require("express");
const Database = require("better-sqlite3");

const db = new Database(":memory:");
db.exec("CREATE TABLE tasks (id INTEGER PRIMARY KEY AUTOINCREMENT, title TEXT, done INTEGER DEFAULT 0)");

const app = express();
app.use(express.json());

app.get("/tasks", (_req, res) =>
  res.json(db.prepare("SELECT * FROM tasks").all())
);
app.post("/tasks", (req, res) => {
  const { title } = req.body;
  const info = db.prepare("INSERT INTO tasks (title) VALUES (?)").run(title);
  res.status(201).json({ id: info.lastInsertRowid, title, done: 0 });
});
app.patch("/tasks/:id", (req, res) => {
  const task = db.prepare("SELECT * FROM tasks WHERE id=?").get(+req.params.id);
  if (!task) return res.status(404).json({ error: "Not found" });
  db.prepare("UPDATE tasks SET done=? WHERE id=?").run(task.done ? 0 : 1, task.id);
  res.json(db.prepare("SELECT * FROM tasks WHERE id=?").get(task.id));
});
app.delete("/tasks/:id", (req, res) => {
  db.prepare("DELETE FROM tasks WHERE id=?").run(+req.params.id);
  res.status(204).end();
});

app.listen(3000, "127.0.0.1", () => console.log("http://127.0.0.1:3000/tasks"));
