Solution 1 — SQLite Note-Taking App

Show solution 
// notes.js — install: npm install better-sqlite3
const Database = require("better-sqlite3");
const path     = require("path");

const db = new Database(path.join(__dirname, "notes.db"));
db.exec(`CREATE TABLE IF NOT EXISTS notes (
  id    INTEGER PRIMARY KEY AUTOINCREMENT,
  text  TEXT NOT NULL,
  done  INTEGER NOT NULL DEFAULT 0,
  ts    TEXT NOT NULL DEFAULT (datetime('now'))
)`);

const [cmd, ...args] = process.argv.slice(2);

if (cmd === "add") {
  const text = args.join(" ");
  if (!text) { console.error("Usage: node notes.js add "); process.exit(1); }
  db.prepare("INSERT INTO notes (text) VALUES (?)").run(text);
  console.log("Note added.");
} else if (cmd === "list") {
  const rows = db.prepare("SELECT * FROM notes ORDER BY id").all();
  rows.forEach(r => console.log(`[${r.id}] [${r.done ? "x" : " "}] ${r.text}  (${r.ts})`));
} else if (cmd === "done") {
  db.prepare("UPDATE notes SET done=1 WHERE id=?").run(+args[0]);
  console.log("Marked done.");
} else if (cmd === "delete") {
  db.prepare("DELETE FROM notes WHERE id=?").run(+args[0]);
  console.log("Deleted.");
} else {
  console.log("Commands: add  | list | done  | delete ");
}

Solution 2 — Repository Pattern for Tasks

Show solution 
// TaskRepository.js
const Database = require("better-sqlite3");

class TaskRepository {
  constructor() {
    this.db = new Database(":memory:");
    this.db.exec(`CREATE TABLE tasks (
      id    INTEGER PRIMARY KEY AUTOINCREMENT,
      title TEXT NOT NULL,
      done  INTEGER NOT NULL DEFAULT 0
    )`);
  }
  findAll()     { return this.db.prepare("SELECT * FROM tasks").all(); }
  findById(id)  { return this.db.prepare("SELECT * FROM tasks WHERE id=?").get(id); }
  create(data)  {
    const info = this.db.prepare("INSERT INTO tasks (title) VALUES (?)").run(data.title);
    return this.findById(info.lastInsertRowid);
  }
  markDone(id)  {
    this.db.prepare("UPDATE tasks SET done=1 WHERE id=?").run(id);
    return this.findById(id);
  }
  delete(id)    { return this.db.prepare("DELETE FROM tasks WHERE id=?").run(id); }
}

// Test script
const repo = new TaskRepository();
console.log(repo.create({ title: "Buy milk" }));
console.log(repo.create({ title: "Write code" }));
console.log(repo.markDone(1));
console.log(repo.findAll());
repo.delete(2);
console.log(repo.findAll());
