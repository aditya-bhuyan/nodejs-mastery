Chapter 7 — Working with Databases Advanced

7.1 SQL vs NoSQL

Feature SQL (e.g. SQLite, PostgreSQL) NoSQL (e.g. MongoDB)
Data modelTables, rows, columnsDocuments (JSON-like)
SchemaStrict, defined upfrontFlexible, evolves easily
JoinsNative SQL JOINs$lookup aggregation
ScalingVertical (stronger machine)Horizontal (more machines)

7.2 SQLite with better-sqlite3

SQLite is a file-based SQL database — perfect for development and small apps.

# Install
npm install better-sqlite3
// db.js
const Database = require("better-sqlite3");
const db = new Database("./app.db");

// Create table
db.exec(`
  CREATE TABLE IF NOT EXISTS users (
    id    INTEGER PRIMARY KEY AUTOINCREMENT,
    name  TEXT NOT NULL,
    email TEXT NOT NULL UNIQUE
  )
`);

// Insert (parameterised — prevents SQL injection)
const insert = db.prepare("INSERT INTO users (name, email) VALUES (?, ?)");
const info   = insert.run("Alice", "alice@example.com");
console.log("Inserted row id:", info.lastInsertRowid);

// Select
const all  = db.prepare("SELECT * FROM users").all();
const one  = db.prepare("SELECT * FROM users WHERE id = ?").get(1);

// Update
db.prepare("UPDATE users SET name = ? WHERE id = ?").run("Alicia", 1);

// Delete
db.prepare("DELETE FROM users WHERE id = ?").run(1);

module.exports = db;

7.3 Transactions

// Wrap multiple writes in a transaction for atomicity
const transfer = db.transaction((fromId, toId, amount) => {
  db.prepare("UPDATE accounts SET balance = balance - ? WHERE id = ?").run(amount, fromId);
  db.prepare("UPDATE accounts SET balance = balance + ? WHERE id = ?").run(amount, toId);
});

transfer(1, 2, 100); // both succeed or both roll back

7.4 MongoDB with Mongoose

# Install
npm install mongoose
// models/User.js
const mongoose = require("mongoose");

const userSchema = new mongoose.Schema({
  name:      { type: String, required: true, trim: true },
  email:     { type: String, required: true, unique: true, lowercase: true },
  createdAt: { type: Date,   default: Date.now },
});

module.exports = mongoose.model("User", userSchema);
// db.js — connect
const mongoose = require("mongoose");

async function connect() {
  const uri = process.env.MONGO_URI || "mongodb://127.0.0.1:27017/myapp";
  await mongoose.connect(uri);
  console.log("MongoDB connected");
}

module.exports = { connect };

7.6 The Repository Pattern

Wrap all DB calls in a repository class to decouple business logic from the database driver:

// UserRepository.js
class UserRepository {
  constructor(db) { this.db = db; }
  findAll()       { return this.db.prepare("SELECT * FROM users").all(); }
  findById(id)    { return this.db.prepare("SELECT * FROM users WHERE id = ?").get(id); }
  create(data)    { return this.db.prepare("INSERT INTO users (name,email) VALUES (?,?)").run(data.name, data.email); }
  delete(id)      { return this.db.prepare("DELETE FROM users WHERE id = ?").run(id); }
}
module.exports = UserRepository;
