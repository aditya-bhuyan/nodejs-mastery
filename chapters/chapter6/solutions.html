Solution 1 — Todo REST API with Express

Show solution 
// todos-api.js  — install: npm install express
const express = require("express");
const app     = express();
app.use(express.json());

const asyncHandler = fn => (req, res, next) =>
  Promise.resolve(fn(req, res, next)).catch(next);

let todos = [], nextId = 1;

const router = express.Router();

router.get("/", asyncHandler(async (req, res) => {
  const { done } = req.query;
  const list = done === undefined ? todos
    : todos.filter(t => t.done === (done === "true"));
  res.json(list);
}));

router.post("/", asyncHandler(async (req, res) => {
  const { title } = req.body;
  if (!title) return res.status(400).json({ error: "title required" });
  const todo = { id: nextId++, title, done: false, createdAt: new Date() };
  todos.push(todo);
  res.status(201).json(todo);
}));

router.put("/:id", asyncHandler(async (req, res) => {
  const idx = todos.findIndex(t => t.id === +req.params.id);
  if (idx === -1) return res.status(404).json({ error: "Not found" });
  todos[idx] = { id: +req.params.id, ...req.body, createdAt: todos[idx].createdAt };
  res.json(todos[idx]);
}));

router.patch("/:id", asyncHandler(async (req, res) => {
  const todo = todos.find(t => t.id === +req.params.id);
  if (!todo) return res.status(404).json({ error: "Not found" });
  todo.done = !todo.done;
  res.json(todo);
}));

router.delete("/:id", asyncHandler(async (req, res) => {
  const idx = todos.findIndex(t => t.id === +req.params.id);
  if (idx === -1) return res.status(404).json({ error: "Not found" });
  todos.splice(idx, 1);
  res.status(204).end();
}));

app.use("/todos", router);
app.use((err, req, res, _next) =>
  res.status(err.status || 500).json({ error: err.message || "Server Error" })
);
app.listen(3000, "127.0.0.1");

Solution 2 — Rate Limiter Middleware

Show solution 
// rateLimiter.js
function rateLimiter(limit, windowMs) {
  const store = new Map(); // ip → [timestamps]
  return function (req, res, next) {
    const ip  = req.socket.remoteAddress;
    const now = Date.now();
    const hits = (store.get(ip) || []).filter(t => now - t < windowMs);
    hits.push(now);
    store.set(ip, hits);
    if (hits.length > limit) {
      return res.status(429).json({ error: "Too Many Requests" });
    }
    next();
  };
}
module.exports = rateLimiter;

// Usage in app.js
const rateLimiter = require("./rateLimiter");
app.use(rateLimiter(100, 60_000)); // 100 req / minute

Solution 3 — JSON Body Validator

Show solution 
// validate.js
function validate(schema) {
  return function (req, res, next) {
    for (const [key, expectedType] of Object.entries(schema)) {
      if (!(key in req.body)) {
        return res.status(400).json({ error: `Missing required field: ${key}` });
      }
      if (typeof req.body[key] !== expectedType) {
        return res.status(400).json({
          error: `Field '${key}' must be of type ${expectedType}`
        });
      }
    }
    next();
  };
}
module.exports = validate;

// Usage
const validate = require("./validate");
app.post("/users", validate({ name: "string", age: "number" }), (req, res) => {
  res.status(201).json(req.body);
});
