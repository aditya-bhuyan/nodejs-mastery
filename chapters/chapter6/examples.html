Chapter 6 — Code Examples

Example 1 — RESTful Users API with Express Router

// users-api/index.js
const express = require("express");
const app     = express();
app.use(express.json());

// --- in-memory store ---
let users  = [{ id: 1, name: "Alice" }, { id: 2, name: "Bob" }];
let nextId = 3;

// --- router ---
const router = express.Router();

router.get("/",    (_req, res) => res.json(users));
router.get("/:id", (req, res) => {
  const user = users.find(u => u.id === +req.params.id);
  user ? res.json(user) : res.status(404).json({ error: "Not found" });
});
router.post("/", (req, res) => {
  const user = { id: nextId++, ...req.body };
  users.push(user);
  res.status(201).json(user);
});
router.put("/:id", (req, res) => {
  const idx = users.findIndex(u => u.id === +req.params.id);
  if (idx === -1) return res.status(404).json({ error: "Not found" });
  users[idx] = { id: +req.params.id, ...req.body };
  res.json(users[idx]);
});
router.delete("/:id", (req, res) => {
  const idx = users.findIndex(u => u.id === +req.params.id);
  if (idx === -1) return res.status(404).json({ error: "Not found" });
  users.splice(idx, 1);
  res.status(204).end();
});

app.use("/users", router);
app.listen(3000, "127.0.0.1", () => console.log("http://127.0.0.1:3000"));

Example 2 — Request Logger Middleware

// logger.js  middleware module
function logger(req, res, next) {
  const start = Date.now();
  res.on("finish", () => {
    console.log(`${req.method} ${req.originalUrl} ${res.statusCode} — ${Date.now()-start}ms`);
  });
  next();
}
module.exports = logger;

// app.js
const logger = require("./logger");
app.use(logger);
