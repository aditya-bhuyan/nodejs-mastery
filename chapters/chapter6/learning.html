Chapter 6 — Express.js Framework Intermediate

6.1 What is Express?

Express is the most widely used Node.js web framework. It wraps the raw http module with a clean API for routing, middleware, and response helpers. Install it:

npm install express

6.2 Your First Express App

// app.js
const express = require("express");
const app     = express();

app.use(express.json());          // parse JSON request bodies
app.use(express.urlencoded({ extended: false })); // parse form bodies

app.get("/", (req, res) => {
  res.json({ message: "Hello, Express!" });
});

const PORT = process.env.PORT || 3000;
app.listen(PORT, "127.0.0.1", () =>
  console.log(`Server running on http://127.0.0.1:${PORT}`)
);

6.3 Routing

// Route parameters
app.get("/users/:id", (req, res) => {
  const { id } = req.params;
  res.json({ id });
});

// Query strings: GET /search?q=node
app.get("/search", (req, res) => {
  const { q } = req.query;
  res.json({ query: q });
});

// Router modules — split routes into separate files
const userRouter = express.Router();
userRouter.get("/",    listUsers);
userRouter.post("/",   createUser);
userRouter.get("/:id", getUser);
userRouter.put("/:id", updateUser);
userRouter.delete("/:id", deleteUser);

app.use("/users", userRouter);

6.4 Middleware

Middleware functions run in a pipeline. Each receives (req, res, next) and must call next() to pass control to the next function.

// Application-level middleware
app.use((req, res, next) => {
  console.log(`${req.method} ${req.url}`);
  next(); // must call next() or the request hangs
});

// Route-level middleware
function authenticate(req, res, next) {
  const token = req.headers.authorization?.split(" ")[1];
  if (!token) return res.status(401).json({ error: "Unauthorized" });
  // verify token…
  next();
}
app.get("/profile", authenticate, (req, res) => {
  res.json({ user: req.user });
});

6.5 Error Handling

Error-handling middleware has four parameters: (err, req, res, next). Register it last.

// Async wrapper — catches thrown errors
function asyncHandler(fn) {
  return (req, res, next) => Promise.resolve(fn(req, res, next)).catch(next);
}

app.get("/data", asyncHandler(async (req, res) => {
  const data = await fetchSomething(); // any throw goes to error handler
  res.json(data);
}));

// Global error handler
app.use((err, req, res, next) => {
  console.error(err.stack);
  res.status(err.status || 500).json({ error: "Internal Server Error" });
});

6.6 Serving Static Files

// Serve everything in ./public at the root URL
app.use(express.static("public"));

// Serve at a sub-path
app.use("/assets", express.static("public"));

6.7 Response Helpers

res.json({ key: "value" });    // 200 with JSON body
res.status(201).json(user);    // custom status + JSON
res.send("plain text");        // text/html by default
res.sendFile(path.join(__dirname, "index.html"));
res.redirect("/new-path");
res.status(404).end();         // no body
