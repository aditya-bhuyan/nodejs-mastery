Chapter 5 — Code Examples

Example 1 — Full In-Memory CRUD REST API

// api.js — run: node api.js
const http = require("http");
const { URL } = require("url");

let users = [
  { id: 1, name: "Alice", email: "alice@example.com" },
  { id: 2, name: "Bob",   email: "bob@example.com"   },
];
let nextId = 3;

function readBody(req) {
  return new Promise((resolve, reject) => {
    let body = "";
    req.on("data", c => (body += c));
    req.on("end",  () => { try { resolve(JSON.parse(body)); } catch { resolve({}); } });
    req.on("error", reject);
  });
}

function send(res, status, data) {
  res.writeHead(status, { "Content-Type": "application/json" });
  res.end(JSON.stringify(data));
}

async function handler(req, res) {
  const { pathname } = new URL(req.url, "http://localhost");
  const parts = pathname.split("/").filter(Boolean); // ["users"] or ["users","1"]
  const resource = parts[0];
  const id = parts[1] ? parseInt(parts[1], 10) : null;

  if (resource !== "users") return send(res, 404, { error: "Not found" });

  if (req.method === "GET" && !id) return send(res, 200, users);

  if (req.method === "GET" && id) {
    const user = users.find(u => u.id === id);
    return user ? send(res, 200, user) : send(res, 404, { error: "User not found" });
  }

  if (req.method === "POST") {
    const body = await readBody(req);
    const user = { id: nextId++, ...body };
    users.push(user);
    return send(res, 201, user);
  }

  if (req.method === "PUT" && id) {
    const idx  = users.findIndex(u => u.id === id);
    if (idx === -1) return send(res, 404, { error: "User not found" });
    const body = await readBody(req);
    users[idx] = { id, ...body };
    return send(res, 200, users[idx]);
  }

  if (req.method === "DELETE" && id) {
    const idx = users.findIndex(u => u.id === id);
    if (idx === -1) return send(res, 404, { error: "User not found" });
    users.splice(idx, 1);
    return send(res, 204, null);
  }

  send(res, 405, { error: "Method Not Allowed" });
}

http.createServer(handler).listen(3000, "127.0.0.1", () =>
  console.log("API running at http://127.0.0.1:3000/users")
);

Example 2 — Query String Parsing

// query-demo.js
const http = require("http");
const { URL } = require("url");

http.createServer((req, res) => {
  const { searchParams } = new URL(req.url, "http://localhost");
  const page  = parseInt(searchParams.get("page")  || "1",  10);
  const limit = parseInt(searchParams.get("limit") || "10", 10);

  res.writeHead(200, { "Content-Type": "application/json" });
  res.end(JSON.stringify({ page, limit, items: [] }));
}).listen(3000, "127.0.0.1");

Example 3 — Consuming an External API with fetch

// fetch-demo.js  (Node.js 18+)
async function getTodos(userId) {
  const res = await fetch(
    `https://jsonplaceholder.typicode.com/todos?userId=${userId}&_limit=5`
  );
  if (!res.ok) throw new Error(`HTTP ${res.status}`);
  return res.json();
}

(async () => {
  const todos = await getTodos(1);
  todos.forEach(t => console.log(`[${t.completed ? "x" : " "}] ${t.title}`));
})();
