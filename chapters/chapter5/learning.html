Chapter 5 — HTTP Server & REST API Intermediate

5.1 The http Module Revisited

Node.js ships with a full HTTP/HTTPS stack. For production APIs you will normally use Express (Chapter 6), but understanding the raw http module helps you debug, write middleware, and understand what frameworks do under the hood.

const http = require("http");

const server = http.createServer((req, res) => {
  res.writeHead(200, { "Content-Type": "application/json" });
  res.end(JSON.stringify({ status: "ok" }));
});

server.listen(3000, "127.0.0.1", () =>
  console.log("Listening on http://127.0.0.1:3000")
);

5.2 The req Object

  • req.method — "GET", "POST", "PUT", "DELETE", …
  • req.url — path + query string, e.g. "/users?page=2"
  • req.headers — all request headers as a plain object
  • req is a Readable stream — read the body by listening to "data" events
function parseBody(req) {
  return new Promise((resolve, reject) => {
    let body = "";
    req.on("data", chunk => (body += chunk));
    req.on("end",  ()    => {
      try { resolve(JSON.parse(body)); }
      catch { resolve({}); }
    });
    req.on("error", reject);
  });
}

5.3 Routing

Without a framework, routing is done by matching req.method and req.url:

const { URL } = require("url");

async function handler(req, res) {
  const { pathname } = new URL(req.url, "http://localhost");

  if (req.method === "GET" && pathname === "/users") {
    return sendJSON(res, 200, users);
  }
  if (req.method === "POST" && pathname === "/users") {
    const body = await parseBody(req);
    const user = { id: Date.now(), ...body };
    users.push(user);
    return sendJSON(res, 201, user);
  }
  sendJSON(res, 404, { error: "Not found" });
}

function sendJSON(res, status, data) {
  res.writeHead(status, { "Content-Type": "application/json" });
  res.end(JSON.stringify(data));
}

5.5 HTTPS

Use the built-in https module for TLS in production. Never serve APIs over plain HTTP.

const https = require("https");
const fs    = require("fs");

const options = {
  key:  fs.readFileSync("./server.key"),
  cert: fs.readFileSync("./server.crt"),
};

https.createServer(options, (req, res) => {
  res.writeHead(200, { "Content-Type": "text/plain" });
  res.end("Secure!");
}).listen(443, "127.0.0.1");

// Native fetch — available in Node.js 18+
