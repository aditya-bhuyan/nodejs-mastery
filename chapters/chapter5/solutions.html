Solution 1 — Query-Filtered List API

Show solution 
// todos-api.js
const http = require("http");
const { URL } = require("url");

const todos = Array.from({ length: 10 }, (_, i) => ({
  id: i + 1, title: `Todo ${i + 1}`, done: i % 2 === 0
}));

function readBody(req) {
  return new Promise((resolve, reject) => {
    let b = "";
    req.on("data", c => (b += c));
    req.on("end", () => { try { resolve(JSON.parse(b)); } catch { resolve({}); } });
    req.on("error", reject);
  });
}
function send(res, status, data) {
  res.writeHead(status, { "Content-Type": "application/json" });
  res.end(data === null ? "" : JSON.stringify(data));
}

async function handler(req, res) {
  const { pathname, searchParams } = new URL(req.url, "http://localhost");
  const parts = pathname.split("/").filter(Boolean);
  if (parts[0] !== "todos") return send(res, 404, { error: "Not found" });
  const id = parts[1] ? parseInt(parts[1], 10) : null;

  if (req.method === "GET" && !id) {
    const filter = searchParams.get("done");
    const result = filter === null ? todos
      : todos.filter(t => t.done === (filter === "true"));
    return send(res, 200, result);
  }
  if (req.method === "GET" && id) {
    const todo = todos.find(t => t.id === id);
    return todo ? send(res, 200, todo) : send(res, 404, { error: "Not found" });
  }
  if (req.method === "PATCH" && id) {
    const todo = todos.find(t => t.id === id);
    if (!todo) return send(res, 404, { error: "Not found" });
    todo.done = !todo.done;
    return send(res, 200, todo);
  }
  send(res, 405, { error: "Method Not Allowed" });
}

http.createServer(handler).listen(3000, "127.0.0.1", () =>
  console.log("Running at http://127.0.0.1:3000/todos")
);

Solution 2 — Request Logger Middleware

Show solution 
// logger-server.js
const http = require("http");

function withLogger(handler) {
  return async function (req, res) {
    const start = Date.now();
    // Intercept writeHead to capture status
    const origWriteHead = res.writeHead.bind(res);
    let statusCode = 200;
    res.writeHead = (status, headers) => {
      statusCode = status;
      return origWriteHead(status, headers);
    };
    await handler(req, res);
    const ms = Date.now() - start;
    console.log(`[${new Date().toISOString()}] ${req.method} ${req.url} → ${statusCode} (${ms}ms)`);
  };
}

function myHandler(req, res) {
  res.writeHead(200, { "Content-Type": "text/plain" });
  res.end("Hello");
}

http.createServer(withLogger(myHandler)).listen(3000, "127.0.0.1");

Solution 3 — Static File Server

Show solution 
// static-server.js
const http = require("http");
const fs   = require("fs");
const path = require("path");

const MIME = {
  ".html": "text/html", ".css": "text/css", ".js": "application/javascript",
  ".json": "application/json", ".png": "image/png", ".jpg": "image/jpeg"
};

http.createServer((req, res) => {
  let filePath = path.join(__dirname, "public", req.url === "/" ? "index.html" : req.url);
  const ext  = path.extname(filePath);
  const mime = MIME[ext] || "application/octet-stream";

  fs.readFile(filePath, (err, data) => {
    if (err) {
      res.writeHead(404, { "Content-Type": "text/plain" });
      return res.end("404 Not Found");
    }
    res.writeHead(200, { "Content-Type": mime });
    res.end(data);
  });
}).listen(3000, "127.0.0.1", () => console.log("Static server at http://127.0.0.1:3000"));
