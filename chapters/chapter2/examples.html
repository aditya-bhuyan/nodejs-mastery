Chapter 2 — Code Examples

Example 1 — Creating and Using a Module

Two files in the same folder.

// utils.js
function capitalize(str) {
  return str.charAt(0).toUpperCase() + str.slice(1);
}

function repeat(str, n) {
  return str.repeat(n);
}

module.exports = { capitalize, repeat };
// app.js
const { capitalize, repeat } = require("./utils");

console.log(capitalize("hello"));    // Hello
console.log(repeat("ha", 3));        // hahaha

Example 2 — Using the path Module

Path operations that work correctly on both Windows (backslash) and Unix (forward slash).

// path-demo.js
const path = require("path");

const fullPath = "/home/user/projects/app/src/index.js";

console.log(path.basename(fullPath));           // index.js
console.log(path.dirname(fullPath));            // /home/user/projects/app/src
console.log(path.extname(fullPath));            // .js
console.log(path.join("src", "utils", "db.js")); // src/utils/db.js (or src\utils\db.js on Windows)
console.log(path.resolve("config", "app.json")); // absolute path

Example 3 — Using the fs Module (sync)

Synchronous file reading — simple but blocks the event loop. Use only in startup scripts, not in servers.

// fs-sync.js
const fs   = require("fs");
const path = require("path");

const filePath = path.join(__dirname, "sample.txt");

// Write a file
fs.writeFileSync(filePath, "Hello, filesystem!\nLine 2\n");

// Read it back
const content = fs.readFileSync(filePath, "utf8");
console.log(content);

// Delete the file
fs.unlinkSync(filePath);

Example 4 — Using the os Module

// os-demo.js
const os = require("os");

console.log("Platform   :", os.platform());   // win32 | darwin | linux
console.log("Arch       :", os.arch());       // x64 | arm64
console.log("Hostname   :", os.hostname());
console.log("Home dir   :", os.homedir());
console.log("CPUs       :", os.cpus().length);
console.log("Free mem   :", (os.freemem() / 1024 ** 3).toFixed(2), "GB");
console.log("Total mem  :", (os.totalmem() / 1024 ** 3).toFixed(2), "GB");

Example 5 — Installing and Using an npm Package (chalk)

First install: npm install chalk@4 (v4 is CommonJS-compatible).

// colors.js
const chalk = require("chalk");

console.log(chalk.green("✓ Success"));
console.log(chalk.red("✗ Error"));
console.log(chalk.yellow.bold("⚠ Warning"));
console.log(chalk.blue(`Running Node.js ${process.version}`));

Run: node colors.js
