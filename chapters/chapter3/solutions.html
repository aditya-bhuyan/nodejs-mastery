Chapter 3 — Solutions

Solution 1 — Promise-based Delay Utility

Show solution 
// delay.js
function delay(ms) {
  return new Promise(resolve => setTimeout(resolve, ms));
}
module.exports = { delay };
// main.js
const { delay } = require("./delay");

async function run() {
  console.log("Step 1");
  await delay(1000);
  console.log("Step 2");
  await delay(500);
  console.log("Step 3 — done");
}

run();

Solution 2 — Async File Copier

Show solution 
// copyfile.js
const fs   = require("fs/promises");
const path = require("path");

async function copyFile(src, dest) {
  const srcAbs  = path.resolve(src);
  const destAbs = path.resolve(dest);

  try {
    const content = await fs.readFile(srcAbs);
    await fs.writeFile(destAbs, content);
    console.log(`Copied ${srcAbs} → ${destAbs}`);
  } catch (err) {
    if (err.code === "ENOENT") {
      console.error(`Error: source file not found — ${srcAbs}`);
    } else {
      console.error("Error:", err.message);
    }
    process.exit(1);
  }
}

const [src, dest] = process.argv.slice(2);
if (!src || !dest) {
  console.error("Usage: node copyfile.js  ");
  process.exit(1);
}

copyFile(src, dest);

Key point: Reading the file as a Buffer (no "utf8" encoding) preserves binary files correctly.

Solution 3 — Parallel vs Sequential Timing

Show solution 
// timing.js
const delay = ms => new Promise(r => setTimeout(r, ms));

async function sequential() {
  const start = Date.now();
  await delay(1000);
  await delay(1000);
  console.log("Sequential elapsed:", Date.now() - start, "ms"); // ~2000 ms
}

async function parallel() {
  const start = Date.now();
  await Promise.all([delay(1000), delay(1000)]);
  console.log("Parallel elapsed  :", Date.now() - start, "ms"); // ~1000 ms
}

async function main() {
  await sequential();
  await parallel();
}

main();

Solution 4 — Event Emitter Logger

Show solution 
// logger.js
const EventEmitter = require("events");

class Logger extends EventEmitter {
  info(msg)  { this.emit("log",   { level: "INFO",  msg }); }
  warn(msg)  { this.emit("log",   { level: "WARN",  msg }); }
  error(msg) { this.emit("error", { level: "ERROR", msg }); }
}

module.exports = Logger;
// app.js
const Logger = require("./logger");
const logger = new Logger();

logger.on("log", ({ level, msg }) => {
  console.log(`[${level}] ${msg}`);
});

logger.on("error", ({ level, msg }) => {
  process.stderr.write(`[${level}] ${msg}\n`);
});

logger.info("Application started");
logger.warn("Low memory warning");
logger.error("Unhandled exception occurred");
