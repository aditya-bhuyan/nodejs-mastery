Chapter 3 — Code Examples

Example 1 — Callback vs Promise vs Async/Await

Three equivalent ways to read a file asynchronously:

// read-callback.js
const fs = require("fs");
fs.readFile("./hello.js", "utf8", (err, data) => {
  if (err) return console.error(err.message);
  console.log("Callback — first 40 chars:", data.slice(0, 40));
});
// read-promise.js
const fs = require("fs").promises;
fs.readFile("./hello.js", "utf8")
  .then(data => console.log("Promise — first 40 chars:", data.slice(0, 40)))
  .catch(err => console.error(err.message));
// read-async.js
const fs = require("fs/promises");

async function readDemo() {
  try {
    const data = await fs.readFile("./hello.js", "utf8");
    console.log("Async/Await — first 40 chars:", data.slice(0, 40));
  } catch (err) {
    console.error(err.message);
  }
}
readDemo();

Example 2 — Creating Your Own Promise

// delay.js
function delay(ms) {
  return new Promise(resolve => setTimeout(resolve, ms));
}

async function run() {
  console.log("Start");
  await delay(1000);
  console.log("After 1 second");
  await delay(500);
  console.log("After another 0.5 seconds");
}

run();

Example 3 — Promise.all for Parallel Operations

// parallel.js
const fs = require("fs/promises");

async function readMultiple() {
  // These two reads happen in parallel, not sequentially
  const [pkg, readme] = await Promise.all([
    fs.readFile("./package.json", "utf8").catch(() => "not found"),
    fs.readFile("./README.md",    "utf8").catch(() => "not found"),
  ]);

  console.log("package.json length:", pkg.length);
  console.log("README.md length   :", readme.length);
}

readMultiple();

Example 4 — Async Error Handling

// async-errors.js
async function fetchData(url) {
  const res = await fetch(url);
  if (!res.ok) throw new Error(`HTTP ${res.status}: ${res.statusText}`);
  return res.json();
}

async function main() {
  try {
    const data = await fetchData("https://jsonplaceholder.typicode.com/todos/1");
    console.log("Got:", data);
  } catch (err) {
    console.error("Request failed:", err.message);
  }
}

main();

Note: fetch is available natively in Node.js 18+.

Example 5 — EventEmitter

// events-demo.js
const EventEmitter = require("events");

class DataPipeline extends EventEmitter {}

const pipeline = new DataPipeline();

// Register listeners
pipeline.on("data",  chunk  => console.log("Received chunk:", chunk));
pipeline.on("end",   ()     => console.log("Pipeline finished."));
pipeline.on("error", err    => console.error("Pipeline error:", err.message));

// Emit events
pipeline.emit("data", { id: 1, value: 42 });
pipeline.emit("data", { id: 2, value: 99 });
pipeline.emit("end");
