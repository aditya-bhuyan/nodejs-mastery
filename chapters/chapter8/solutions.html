Solution 1 — Secure Register/Login API

Show solution 
// auth-api.js — npm install express bcrypt jsonwebtoken dotenv
require("dotenv").config();
const express = require("express");
const bcrypt  = require("bcrypt");
const jwt     = require("jsonwebtoken");

const SECRET = process.env.JWT_SECRET;
if (!SECRET) throw new Error("JWT_SECRET must be set in environment");

const app   = express();
const users = [];
app.use(express.json());

app.post("/register", async (req, res) => {
  const { email, password } = req.body;
  if (!email || !password || password.length < 8)
    return res.status(400).json({ error: "email required; password min 8 chars" });
  if (users.find(u => u.email === email))
    return res.status(409).json({ error: "Email already in use" });
  const hash = await bcrypt.hash(password, 12);
  const user = { id: users.length + 1, email, hash };
  users.push(user);
  res.status(201).json({ id: user.id, email: user.email });
});

app.post("/login", async (req, res) => {
  const { email, password } = req.body;
  const user = users.find(u => u.email === email);
  if (!user || !(await bcrypt.compare(password, user.hash)))
    return res.status(401).json({ error: "Invalid credentials" });
  const token = jwt.sign({ sub: user.id, email: user.email }, SECRET, { expiresIn: "1h" });
  res.json({ token });
});

function auth(req, res, next) {
  const [, token] = (req.headers.authorization || "").split(" ");
  if (!token) return res.status(401).json({ error: "No token provided" });
  try { req.user = jwt.verify(token, SECRET); next(); }
  catch { res.status(401).json({ error: "Invalid or expired token" }); }
}

app.get("/me", auth, (req, res) =>
  res.json({ id: req.user.sub, email: req.user.email })
);

app.listen(3000, "127.0.0.1");

Solution 2 — Joi Validation Middleware

Show solution 
// validate.js — npm install joi
const Joi = require("joi");

function validate(schema) {
  return (req, res, next) => {
    const { error, value } = schema.validate(req.body, { abortEarly: false });
    if (error) {
      const messages = error.details.map(d => d.message);
      return res.status(400).json({ errors: messages });
    }
    req.body = value; // use the validated/coerced value
    next();
  };
}
module.exports = validate;

// Usage in app.js
const Joi      = require("joi");
const validate = require("./validate");

const productSchema = Joi.object({
  name:     Joi.string().min(2).max(100).required(),
  price:    Joi.number().positive().required(),
  category: Joi.string().required(),
});

app.post("/products", validate(productSchema), (req, res) =>
  res.status(201).json(req.body)
);
