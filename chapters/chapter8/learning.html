Chapter 8 — Authentication & Security Advanced

8.1 Why Security Matters

Node.js APIs are high-value attack targets. The OWASP Top 10 categories most relevant to Node.js APIs are: Injection, Broken Authentication, Sensitive Data Exposure, Security Misconfiguration, and Insufficient Logging. Addressing these proactively is far cheaper than incident response.

8.2 Hashing Passwords with bcrypt

Never store plaintext passwords. Use an adaptive hashing algorithm like bcrypt:

npm install bcrypt
const bcrypt = require("bcrypt");
const SALT_ROUNDS = 12; // higher = slower hash = harder to brute-force

// Hash on registration
async function hashPassword(plain) {
  return bcrypt.hash(plain, SALT_ROUNDS);
}

// Verify on login
async function verifyPassword(plain, hashed) {
  return bcrypt.compare(plain, hashed);
}

// Example
const hash = await hashPassword("myS3cr3t!");
console.log(await verifyPassword("myS3cr3t!", hash));  // true
console.log(await verifyPassword("wrong",     hash));  // false

8.3 JSON Web Tokens (JWT)

JWT is the standard for stateless API authentication. A token consists of three base64-encoded parts: header.payload.signature.

npm install jsonwebtoken
const jwt = require("jsonwebtoken");
const SECRET = process.env.JWT_SECRET; // NEVER hardcode; load from env

// Issue token on successful login
function issueToken(userId) {
  return jwt.sign({ sub: userId }, SECRET, { expiresIn: "1h" });
}

// Verify token in middleware
function authenticate(req, res, next) {
  const auth = req.headers.authorization;
  if (!auth?.startsWith("Bearer ")) {
    return res.status(401).json({ error: "Missing token" });
  }
  try {
    const payload = jwt.verify(auth.split(" ")[1], SECRET);
    req.user = payload;
    next();
  } catch {
    res.status(401).json({ error: "Invalid or expired token" });
  }
}

8.5 Rate Limiting

npm install express-rate-limit
const rateLimit = require("express-rate-limit");

const limiter = rateLimit({
  windowMs: 15 * 60 * 1000, // 15 minutes
  max: 100,                  // max 100 requests per window per IP
  standardHeaders: true,
  legacyHeaders: false,
  message: { error: "Too many requests, please try again later." },
});

app.use("/api/", limiter); // apply to all /api/ routes

8.6 Input Validation

npm install joi
const Joi = require("joi");

const registerSchema = Joi.object({
  name:     Joi.string().min(2).max(50).required(),
  email:    Joi.string().email().required(),
  password: Joi.string().min(8).pattern(/[A-Z]/).pattern(/[0-9]/).required(),
});

function validateRegister(req, res, next) {
  const { error } = registerSchema.validate(req.body);
  if (error) return res.status(400).json({ error: error.details[0].message });
  next();
}

app.post("/register", validateRegister, registerHandler);
