Chapter 9 — Testing & Debugging Advanced

9.1 Why Test?

Automated tests catch regressions, document expected behaviour, and give you confidence to refactor. Node.js projects typically have three layers of tests:

  • Unit tests — test a single function or class in isolation
  • Integration tests — test how several modules work together (e.g. route + database)
  • End-to-end tests — test the complete running application through its public API

9.2 Jest — Setup

npm install --save-dev jest

Add to package.json:

"scripts": {
  "test":       "jest",
  "test:watch": "jest --watch",
  "test:cover": "jest --coverage"
}

Jest automatically discovers files matching **/*.test.js or **/*.spec.js.

9.3 Writing Unit Tests

// math.js
function add(a, b) { return a + b; }
function divide(a, b) {
  if (b === 0) throw new Error("Division by zero");
  return a / b;
}
module.exports = { add, divide };

// math.test.js
const { add, divide } = require("./math");

describe("add()", () => {
  it("adds two positive numbers", () => {
    expect(add(2, 3)).toBe(5);
  });
  it("handles negatives", () => {
    expect(add(-1, 1)).toBe(0);
  });
});

describe("divide()", () => {
  it("divides correctly", () => {
    expect(divide(10, 2)).toBe(5);
  });
  it("throws on division by zero", () => {
    expect(() => divide(5, 0)).toThrow("Division by zero");
  });
});

9.4 Testing Async Code

// async.test.js
const { delay } = require("./delay");

test("delay resolves after the specified ms", async () => {
  const start = Date.now();
  await delay(100);
  expect(Date.now() - start).toBeGreaterThanOrEqual(90);
});

// Testing rejected promises
test("divide rejects on bad input", async () => {
  await expect(Promise.reject(new Error("fail"))).rejects.toThrow("fail");
});

9.5 Mocking with Jest

// userService.test.js
const UserService = require("./userService");
const db          = require("./db");

jest.mock("./db"); // auto-mock the module

test("getUser returns user from database", async () => {
  db.findById.mockResolvedValue({ id: 1, name: "Alice" });

  const user = await UserService.getUser(1);
  expect(user.name).toBe("Alice");
  expect(db.findById).toHaveBeenCalledWith(1);
});

9.6 Integration Testing Express Routes

npm install --save-dev supertest
// app.test.js
const request = require("supertest");
const app     = require("./app"); // your Express app (don't call .listen())

describe("GET /users", () => {
  it("returns 200 and an array", async () => {
    const res = await request(app).get("/users");
    expect(res.status).toBe(200);
    expect(Array.isArray(res.body)).toBe(true);
  });
});

describe("POST /users", () => {
  it("creates a user and returns 201", async () => {
    const res = await request(app)
      .post("/users")
      .send({ name: "Charlie", email: "charlie@example.com" });
    expect(res.status).toBe(201);
    expect(res.body).toHaveProperty("id");
  });
});

9.7 Debugging

The --inspect Flag

# Start app in debug mode
node --inspect app.js

# Open Chrome: chrome://inspect → click "Open dedicated DevTools for Node"

VS Code Debugging

Add a .vscode/launch.json:

{
  "version": "0.2.0",
  "configurations": [{
    "type": "node",
    "request": "launch",
    "name": "Debug App",
    "program": "${workspaceFolder}/app.js",
    "envFile": "${workspaceFolder}/.env"
  }]
}

Set breakpoints in the editor, press F5 to start debugging.

9.8 The console Debugging Toolkit

console.log("value:", value);             // basic
console.dir(obj, { depth: null });        // deep-inspect object
console.time("label"); /* ... */ console.timeEnd("label");  // timing
console.table(arrayOfObjects);            // tabular output
console.trace("reached here");           // prints call stack
