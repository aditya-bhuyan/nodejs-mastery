Solution 1 — Unit Test a String Utility Module

Show solution 
// stringUtils.test.js
const { capitalize, reverseWords, isPalindrome, countWords } = require("./stringUtils");

describe("capitalize()", () => {
  it("capitalises first letter",    () => expect(capitalize("hello")).toBe("Hello"));
  it("handles all-caps",            () => expect(capitalize("WORLD")).toBe("WORLD"));
  it("handles empty string",        () => expect(capitalize("")).toBe(""));
});

describe("reverseWords()", () => {
  it("reverses multiple words",     () => expect(reverseWords("foo bar baz")).toBe("baz bar foo"));
  it("single word unchanged",       () => expect(reverseWords("hello")).toBe("hello"));
});

describe("isPalindrome()", () => {
  it("detects palindrome",          () => expect(isPalindrome("racecar")).toBe(true));
  it("case-insensitive",            () => expect(isPalindrome("RaceCar")).toBe(true));
  it("non-palindrome",              () => expect(isPalindrome("hello")).toBe(false));
  it("multi-word palindrome",       () => expect(isPalindrome("A man a plan a canal Panama")).toBe(true));
});

describe("countWords()", () => {
  it("counts words correctly",      () => expect(countWords("hello world")).toBe(2));
  it("handles extra spaces",        () => expect(countWords("  foo   bar  ")).toBe(2));
  it("single word",                 () => expect(countWords("node")).toBe(1));
});

Solution 2 — Integration Test a REST API

Show solution 
// todos.test.js — npm install --save-dev jest supertest
const request = require("supertest");
const app     = require("./app"); // export app without .listen()

describe("Todos API", () => {
  let createdId;

  test("GET /todos → 200 array", async () => {
    const res = await request(app).get("/todos");
    expect(res.status).toBe(200);
    expect(Array.isArray(res.body)).toBe(true);
  });

  test("POST /todos → 201 with id", async () => {
    const res = await request(app).post("/todos").send({ title: "Test todo" });
    expect(res.status).toBe(201);
    expect(res.body).toHaveProperty("id");
    createdId = res.body.id;
  });

  test("POST /todos missing title → 400", async () => {
    const res = await request(app).post("/todos").send({});
    expect(res.status).toBe(400);
  });

  test("PATCH /todos/:id toggles done", async () => {
    const res = await request(app).patch(`/todos/${createdId}`);
    expect(res.status).toBe(200);
    expect(res.body.done).toBe(true);
  });

  test("DELETE /todos/:id → 204", async () => {
    const res = await request(app).delete(`/todos/${createdId}`);
    expect(res.status).toBe(204);
  });

  test("GET /todos/:id not found → 404", async () => {
    const res = await request(app).get("/todos/99999");
    expect(res.status).toBe(404);
  });
});

Solution 3 — Mock an External API Call

Show solution 
// weather.js
async function getWeather(city) {
  const res  = await fetch(`https://wttr.in/${encodeURIComponent(city)}?format=j1`);
  const data = await res.json();
  return data.current_condition[0].temp_C;
}
module.exports = { getWeather };

// weather.test.js
const { getWeather } = require("./weather");

beforeEach(() => {
  global.fetch = jest.fn().mockResolvedValue({
    ok: true,
    json: async () => ({
      current_condition: [{ temp_C: "15" }]
    })
  });
});

afterEach(() => jest.restoreAllMocks());

test("getWeather returns temperature for London", async () => {
  const temp = await getWeather("London");
  expect(temp).toBe("15");
  expect(global.fetch).toHaveBeenCalledWith(
    "https://wttr.in/London?format=j1"
  );
});
