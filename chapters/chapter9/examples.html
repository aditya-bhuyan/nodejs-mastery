Chapter 9 — Code Examples

Example 1 — Unit Tests for a Calculator

// calc.test.js — run: npx jest calc.test.js
const calc = {
  add:      (a, b) => a + b,
  subtract: (a, b) => a - b,
  multiply: (a, b) => a * b,
  divide(a, b) {
    if (b === 0) throw new Error("Division by zero");
    return a / b;
  }
};

describe("Calculator", () => {
  test("add(3, 4) === 7",      () => expect(calc.add(3, 4)).toBe(7));
  test("subtract(10, 3) === 7",() => expect(calc.subtract(10, 3)).toBe(7));
  test("multiply(3, 4) === 12",() => expect(calc.multiply(3, 4)).toBe(12));
  test("divide(10, 2) === 5",  () => expect(calc.divide(10, 2)).toBe(5));
  test("divide by zero throws",() => {
    expect(() => calc.divide(1, 0)).toThrow("Division by zero");
  });
});

Example 2 — Integration Test with Supertest

// app.js — simple Express app to test
const express = require("express");
const app     = express();
app.use(express.json());

const items = [{ id: 1, name: "apple" }];
app.get("/items",     (_req, res) => res.json(items));
app.post("/items",    (req,  res) => {
  const item = { id: items.length + 1, ...req.body };
  items.push(item);
  res.status(201).json(item);
});

module.exports = app; // export WITHOUT calling .listen()

// app.test.js — install: npm i -D jest supertest
const request = require("supertest");
const app     = require("./app");

test("GET /items returns array", async () => {
  const res = await request(app).get("/items");
  expect(res.statusCode).toBe(200);
  expect(Array.isArray(res.body)).toBe(true);
});

test("POST /items creates item", async () => {
  const res = await request(app).post("/items").send({ name: "banana" });
  expect(res.statusCode).toBe(201);
  expect(res.body.name).toBe("banana");
  expect(res.body).toHaveProperty("id");
});

Example 3 — Mocking a Dependency

// emailService.js
async function sendWelcomeEmail(to) {
  // would call an external SMTP service
  return { accepted: [to] };
}
module.exports = { sendWelcomeEmail };

// userService.js
const email = require("./emailService");
async function register(user) {
  // ...save user to DB...
  await email.sendWelcomeEmail(user.email);
  return user;
}
module.exports = { register };

// userService.test.js
jest.mock("./emailService");
const email       = require("./emailService");
const userService = require("./userService");

test("register sends welcome email", async () => {
  email.sendWelcomeEmail.mockResolvedValue({ accepted: ["a@b.com"] });
  const user = await userService.register({ name: "Alice", email: "a@b.com" });
  expect(email.sendWelcomeEmail).toHaveBeenCalledWith("a@b.com");
  expect(user.name).toBe("Alice");
});
